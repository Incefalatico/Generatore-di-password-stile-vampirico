cmake_minimum_required(VERSION 3.16)

project(PasswordGenerator LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Find Qt package (Qt6 or Qt5)
find_package(Qt6 REQUIRED COMPONENTS Widgets)

add_executable(PasswordGenerator
    "password generator.cpp"
)

target_link_libraries(PasswordGenerator PRIVATE Qt6::Widgets)

# Handle Windows subsystem (avoids opening a console window alongside the GUI)
if(WIN32)
    set_target_properties(PasswordGenerator PROPERTIES WIN32_EXECUTABLE ON)
endif()
